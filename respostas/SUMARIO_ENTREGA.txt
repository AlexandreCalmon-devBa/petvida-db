1️⃣  fn_idade_animal(data_nascimento) → VARCHAR
    ✅ Retorna: "X anos e Y meses"
    ✅ Usa: TIMESTAMPDIFF
    ✅ Exemplo: fn_idade_animal(data_nascimento) = "5 anos e 3 meses"

2️⃣  fn_total_gasto_tutor(tutor_id) → DECIMAL
    ✅ Retorna: Soma de consultas (exclui canceladas)
    ✅ Usa: SUM com INNER JOIN e WHERE
    ✅ Exemplo: fn_total_gasto_tutor(1) = 2030.00

3️⃣  fn_qtd_consultas_animal(animal_id) → INT
    ✅ Retorna: Quantidade total de consultas
    ✅ Usa: COUNT
    ✅ Exemplo: fn_qtd_consultas_animal(1) = 2

4️⃣  fn_status_emoji(status) → VARCHAR
    ✅ Retorna: Status com emoji descritivo
    ✅ Mapeia: 📅 ✅ 🏥 ❌
    ✅ Exemplo: fn_status_emoji('concluida') = "✅ Concluída"

5️⃣  fn_classificar_valor(valor) → VARCHAR
    ✅ Retorna: Classificação do procedimento
    ✅ Classifica: Simples | Padrão | Especial
    ✅ Exemplo: fn_classificar_valor(250) = "Consulta Padrão"
